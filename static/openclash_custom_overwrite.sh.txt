begin
  cfg = YAML.load_file(ENV["CONFIG_FILE"])

  names = cfg["proxies"].map { |p| p["name"] }
  if names.size == 0
    puts "订阅转换: proxies 为空(可能 provider 模式), 跳过注入以免生成空组"
  else
    countries = YAML.load(ENV["OC_COUNTRY"])
    find = lambda { |nm|
      lower = nm.to_s.downcase
      toks = nm.to_s.upcase.split(/[^A-Z0-9]+/).reject { |x| x.empty? }
      res = nil
      countries.each { |r|
        r["keywords"].each { |k|
          if k =~ /\A[A-Za-z]{2,3}\z/
            if toks.include?(k.upcase) then res = r["name"]; break end
          else
            if lower.include?(k.downcase) then res = r["name"]; break end
          end
        }
        break if res
      }
      res
    }
    cmap = {}
    others = []
    names.each { |n|
      hit = find.call(n)
      if hit then (cmap[hit] ||= []) << n else others << n end
    }
    active = countries.map { |c| c["name"] }.select { |nm| (cmap[nm] || []).size > 0 }
    url = "http://cp.cloudflare.com/generate_204"
    g = []
    g << { "name" => "手动选择", "type" => "select", "proxies" => (active + ["♻️自动选择", "DIRECT"]), "include-all-proxies" => true }
    g << { "name" => "♻️自动选择", "type" => "url-test", "url" => url, "interval" => 300, "tolerance" => 50, "proxies" => names }
    ["AIGC", "PanDownload", "PicHub", "Telegram", "API中转站", "Google", "🐟漏网之鱼"].each { |nm|
      g << { "name" => nm, "type" => "select", "proxies" => (["手动选择"] + active + ["♻️自动选择", "DIRECT"]), "include-all-proxies" => true }
    }
    g << { "name" => "🛑ADblock", "type" => "select", "proxies" => ["REJECT", "DIRECT"] }
    g << { "name" => "Adobe", "type" => "select", "proxies" => ["REJECT", "DIRECT"] }
    active.each { |nm|
      g << { "name" => nm, "type" => "url-test", "url" => url, "interval" => 300, "tolerance" => 50, "proxies" => cmap[nm] }
    }
    if others.size > 0
      g << { "name" => "🌐其他 Auto", "type" => "url-test", "url" => url, "interval" => 300, "tolerance" => 50, "proxies" => others }
    end
    cfg["proxy-groups"] = g
    cfg["rule-providers"] = YAML.load(ENV["OC_RPROV"])
    cfg["rules"] = YAML.load(ENV["OC_RULES"])
    cfg["dns"] = YAML.load(ENV["OC_DNS"])
    cfg["port"] = 7890
    cfg["socks-port"] = 7891
    cfg["allow-lan"] = true
    cfg["ipv6"] = false
    cfg["bind-address"] = "*"
    cfg["log-level"] = "info"
    cfg["external-controller"] = "0.0.0.0:9090"
    cfg["unified-delay"] = true
    cfg["tcp-concurrent"] = true
    cfg["skip-auth-prefixes"] = ["192.168.1.0/24", "192.168.31.0/24", "192.168.100.0/24", "127.0.0.1/8"]
    File.open(ENV["CONFIG_FILE"], "w") { |f| YAML.dump(cfg, f) }
    puts "订阅转换规则注入成功: 国家组 #{active.size} 个, 未归类 #{others.size} 个, 节点 #{names.size} 个"
  end
rescue => e
  puts "订阅转换规则注入失败: #{e.message}"
end
'
